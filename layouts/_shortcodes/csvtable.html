{{- /* csvtable: render a CSV asset as a styled, horizontally scrollable schedule grid. Expected columns: Session, Time, Presentation, Authors, PDF, arXiv. Repeated Session labels are blanked so each session reads as a group. */ -}} {{- $src := .Get "src" -}} {{- $res := resources.Get $src -}} {{- if not $res -}} {{- errorf "csvtable: resource %q not found" $src -}} {{- end -}} {{- $rows := $res | transform.Unmarshal -}}
{{- $prev := "" -}} {{- range after 1 $rows -}} {{- $session := index . 0 -}} {{- $time := index . 1 -}} {{- $title := index . 2 -}} {{- $authors := index . 3 -}} {{- $pdf := "" -}}{{- if ge (len .) 5 }}{{ $pdf = index . 4 }}{{ end -}} {{- $arxiv := "" -}}{{- if ge (len .) 6 }}{{ $arxiv = index . 5 }}{{ end -}} {{- $prev = $session -}} {{- end -}}
